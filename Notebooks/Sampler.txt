def SamplerFoil(contour, step):
    """Генерируем точки на контуре с заданным шагом"""
    line = LineString(contour)
    length = line.length
    points = []
    for distance in np.arange(0, length, step):
        point = line.interpolate(distance)
        points.append((point.x, point.y))
        
    return points
def Sampler(foil, domain, step):
    poly_foil = Polygon(foil)
    poly_domain = Polygon(domain)
    area = poly_domain.difference(poly_foil)
    minx, miny, maxx, maxy = area.bounds
    points = []
    Number_of_points = (maxx-minx)*(maxy-miny)/step**2

    for x in np.arange(minx, maxx, step):
        for y in np.arange(miny, maxy, step):
            point = Point(x, y)
            Number_of_points = Number_of_points - 1.0
            print("Точек осталось: ", Number_of_points)
            if area.contains(point):
                points.append((x, y))
    return points

def Plotter(outer_contour, inner_contour, points):
    plt.figure(figsize=(8, 8))
    # Внешний контур
    xs, ys = zip(*outer_contour)
    plt.plot(xs + (xs[0],), ys + (ys[0],), 'b-', label='Внешний контур')
    # Внутренний контур
    xs, ys = zip(*inner_contour)
    plt.plot(xs + (xs[0],), ys + (ys[0],), 'r--', label='Внутренний контур')
    # Точки
    px, py = zip(*points)
    plt.scatter(px, py, color='green', s=5, label='Сгенерированные точки')   
    plt.legend()
    plt.axis('equal')
    plt.grid(True)
    plt.title("Генерация точек между контурами")
    plt.show()

def Saving(coordinates, filename):
    # Проверяем формат входных данных
    if not all(isinstance(coord, tuple) and len(coord) == 2 for coord in coordinates):
        raise ValueError("Координаты должны быть списком кортежей с двумя элементами")
    
    # Разделяем координаты на x и y
    x_coords = [str(point[0]) for point in coordinates]
    y_coords = [str(point[1]) for point in coordinates]
    
    # Открываем файл для записи
    with open(filename, 'w', encoding='utf-8') as file:
        # Записываем заголовок
        #file.write("X\tY\n")
        # Записываем данные построчно
        for x, y in zip(x_coords, y_coords):
            file.write(f"{x}\t{y}\n")
