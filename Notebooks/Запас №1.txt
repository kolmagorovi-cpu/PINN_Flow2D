def read_coordinates_from_file(file_path):
    """
    Функция для чтения координат из текстового файла
    Параметры:
    file_path (str): путь к файлу с координатами
    Возвращает:
    list: список кортежей с координатами (x, y)
    """
    coordinates = []
    
    try:
        # Открываем файл в режиме чтения
        with open(file_path, 'r') as file:
            # Проходим по каждой строке файла
            for line_number, line in enumerate(file, start=1):
                # Удаляем пробелы и переносы строк
                line = line.strip()
                
                # Пропускаем пустые строки
                if not line:
                    continue
                #print(line.split('\t'))
                # Разделяем значения по табуляции
                try:
                    x_str, y_str, z_str = line.split('\t')
                    x = float(x_str)  # Преобразуем в число
                    y = float(y_str)  # Преобразуем в число
                    z = float(z_str)  # Преобразуем в число
                    coordinates.append((x, y))
                except ValueError:
                    print(f"Ошибка в строке {line_number}: некорректный формат данных")
                    
    except FileNotFoundError:
        print(f"Ошибка: файл по пути {file_path} не найден")
    except Exception as e:
        print(f"Произошла ошибка при чтении файла: {str(e)}")
        
    return coordinates

import matplotlib.pyplot as plt

def plot_coordinates(coordinates, title='График координат', 
                    x_label='X', y_label='Y', 
                    show_grid=True, 
                    marker='o', 
                    line_style='-', 
                    color='blue'):
    """
    Функция для построения графика по координатам
    Параметры:
    coordinates (list of tuples): список кортежей с координатами (x, y)
    title (str): заголовок графика
    x_label (str): подпись оси X
    y_label (str): подпись оси Y
    show_grid (bool): показывать сетку
    marker (str): тип маркера точек
    line_style (str): стиль линии
    color (str): цвет графика
    """
    # Разделяем координаты на X и Y
    x = [coord[0] for coord in coordinates]
    y = [coord[1] for coord in coordinates]
    
    # Создаем график
    plt.figure(figsize=(10, 6))
    
    # Строим линию
    plt.plot(x, y, 
             linestyle=line_style, 
             marker=marker, 
             color=color, 
             label='Координаты')
    
    # Добавляем точки
    plt.scatter(x, y, color=color)
    
    # Настраиваем внешний вид
    plt.title(title)
    plt.xlabel(x_label)
    plt.ylabel(y_label)
    
    if show_grid:
        plt.grid(True)
        
    plt.legend()
    plt.tight_layout()  # Автоматическое расположение элементов
    plt.show()
def countour_closure_condition(coordinates):
    ccc = coordinates [0] == coordinates[-1]
    return abs(-ccc)
