Структура проекта
=================

pinn_flow_2d/                  # Корневая папка проекта
├── source/                      # Исходный код (основной пакет)
│   ├── __init__.py
│   ├── models/               # Реализация PINN-архитектур
│   │   ├── __init__.py
│   │   ├── base_pinn.py      # Базовый класс PINN
│   │   ├── navier_stokes_pinn.py  # PINN для уравнений Навье‑Стокса
│   │   └── potential_flow_pinn.py # PINN для потенциального течения
│   ├── solvers/              # Решатели и алгоритмы
│   │   ├── __init__.py
│   │   ├── pinn_solver.py     # Основной решатель PINN
│   │   └── optimizer.py       # Настройки оптимизаторов
│   ├── physics/              # Физические законы и уравнения
│   │   ├── __init__.py
│   │   ├── equations.py       # Уравнения гидродинамики (Навье‑Стокса, Эйлера)
│   │   └── boundary_conditions.py  # Граничные условия
│   ├── data/                 # Работа с данными
│   │   ├── __init__.py
│   │   ├── dataset.py          # Класс датасета для PINN
│   │   └── preprocessor.py     # Препроцессинг данных
│   └── utils/                # Вспомогательные функции
│       ├── __init__.py
│       ├── visualization.py    # Визуализация результатов
│       └── logger.py         # Логирование экспериментов
├── configs/                  # Конфигурации и гиперпараметры
│   ├── default_config.yaml    # Базовая конфигурация
│   ├── ns_config.yaml         # Конфигурация для Навье‑Стокса
│   └── potential_config.yaml   # Конфигурация для потенциального течения
├── experiments/              # Эксперименты и запуски
│   ├── __init__.py
│   ├── run_training.py         # Скрипт запуска обучения
│   ├── analyze_results.py     # Анализ результатов
│   └── experiment_tracker.py  # Трекер экспериментов
├── notebooks/                # Jupyter‑ноутбуки для исследований
│   ├── data_exploration.ipynb  # Анализ данных
│   ├── model_prototyping.ipynb # Прототипирование моделей
│   └── results_visualization.ipynb  # Визуализация
├── data/                     # Данные (вне репозитория, в .gitignore)
│   ├── raw/                  # Сырые данные (например, сетки, начальные условия)
│   ├── processed/            # Обработанные данные для обучения
│   └── simulations/          # Результаты численных симуляций (для валидации)
├── models/                   # Сохранённые модели и чекпоинты
│   ├── checkpoints/            # Чекпоинты обучения
│   └── final_models/         # Итоговые обученные модели
├── logs/                     # Логи обучения и метрики
│   ├── training/               # Логи тренировок
│   └── metrics/              # Метрики и кривые обучения
├── results/                  # Итоговые результаты
│   ├── figures/              # Графики и визуализации
│   ├── tables/               # Таблицы с метриками
│   └── reports/              # Отчёты по экспериментам
├── tests/                    # Тесты
│   ├── __init__.py
│   ├── test_models.py          # Тесты моделей
│   ├── test_solvers.py        # Тесты решателей
│   └── test_data.py          # Тесты работы с данными
├── requirements.txt          # Зависимости проекта
├── setup.py                  # Сборка пакета (опционально)
├── README.md                 # Описание проекта
└── .gitignore                # Игнорируемые файлы
